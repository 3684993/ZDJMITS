import fs from 'node:fs';
import {DatabaseSync} from 'node:sqlite';
const root='D:/MITS',out=process.env.TEMP+'/mits-v397-final-audit';
const parse=x=>{try{return typeof x==='string'?JSON.parse(x):x}catch{return null}};
const db=new DatabaseSync(root+'/data/zdj-settings.sqlite',{readOnly:true});db.exec('PRAGMA query_only=ON');
const now=Date.now(),identity=parse(fs.readFileSync(root+'/data/runtime/engine-instance.json','utf8')),since=identity.startedAt;
const entities=k=>db.prepare('select payload from runtime_entities where kind=?').all(k).map(x=>parse(x.payload)).filter(Boolean);
const orders=entities('entryOrders'),intents=entities('entryIntents'),plans=entities('tradePlans'),reserves=entities('entryReservations'),positions=entities('positions'),fills=entities('executionFills'),records=db.prepare('select payload from trade_records').all().map(x=>parse(x.payload));
const runs=db.prepare("select payload from ai_runs_archive where role='PRIMARY_BRAIN' and started_at>=? order by started_at").all(now-86400000).map(x=>parse(x.payload));
const events=db.prepare("select type,ts,symbol,payload from runtime_events where ts>=? and (type like 'ENTRY_%' or type like 'AI_SIZING%' or type like 'TRADE_PLAN%' or type like 'CANDIDATE_REJECTED' or type='FINAL_ORDER_RISK_EVALUATED' or type='PORTFOLIO_RISK_ADMISSION_EVALUATED' or type like 'TP_%' or type like 'AI_EXIT_%') order by ts").all(since).map(x=>({...x,payload:parse(x.payload)}));
const stats=a=>{a=a.filter(Number.isFinite).sort((a,b)=>a-b);const q=x=>a.length?a[Math.floor((a.length-1)*x)]:null;return{n:a.length,sum:a.reduce((a,b)=>a+b,0),min:q(0),p10:q(.1),p50:q(.5),p90:q(.9),max:q(1)}};
const countBy=(a,f)=>{const r={};for(const x of a){const k=f(x);r[k]=(r[k]??0)+1;}return r};
const cohort=t=>{const a=orders.filter(x=>x.createdAt>=t),b=fills.filter(x=>x.executionTime>=t);return{orders:a.length,notional:stats(a.map(x=>x.quantity*x.price)),notionalBuckets:countBy(a,x=>{const n=x.quantity*x.price;return n<4.9?'<4.9':n<=5.5?'4.9-5.5':n<20?'5.5-20':n<200?'20-200':'>=200'}),bySide:countBy(a,x=>x.side),byQuote:countBy(a,x=>x.symbol.endsWith('USDC')?'USDC':'USDT'),byStatus:countBy(a,x=>x.status),entryFillCount:b.filter(x=>x.fillRole==='ENTRY'||(!x.fillRole&&x.side===(x.direction==='LONG'?'BUY':'SELL'))).length,exitFillCount:b.filter(x=>x.fillRole==='EXIT'||(!x.fillRole&&x.side!==(x.direction==='LONG'?'BUY':'SELL'))).length,closedCycles:records.filter(x=>x.status==='CLOSED'&&x.closedAt>=t&&!x.duplicateOf&&x.canonical!==false).length}};
function trace(r){const p=parse(r.inputPreview)?.packet,rawOuter=parse(r.outputPreview);let raw=parse(rawOuter?.choices?.[0]?.message?.content);const normalized=parse(r.normalizedPreview);const i=intents.find(x=>x.brainRunId===r.id),plan=plans.find(x=>x.planId===i?.planId)||plans.find(x=>x.provenance?.modelRunId===r.id),reservation=reserves.find(x=>x.id===i?.reservationId),os=orders.filter(x=>x.intentId===i?.id);const e=p?.executionEnvelope,side=normalized?.tradeSide??r.direction,step=e?.exchange?.stepSize;const ev=events.filter(x=>x.payload?.brainRunId===r.id||x.payload?.intentId===i?.id&&i||x.payload?.intent?.brainRunId===r.id);return{runId:r.id,symbol:r.symbol,startedAt:r.startedAt,completedAt:r.completedAt,status:r.status,input:{packetId:p?.packetId,createdAt:p?.createdAt,quoteAt:p?.market?.quote?.ts,quote:p?.market?.quote,executionEnvelope:e,capitalEnvelope:p?.capitalEnvelope,selection:p?.selection,scoutHandoff:r.scoutHandoff},raw,normalized,plan,intent:i,reservation,orders:os,events:ev,comparison:{rawUnits:raw?.quantityUnits,normalizedUnits:normalized?.quantityUnits,stepSize:step,rawQty:raw?.quantityUnits*step,planUnits:plan?.quantityUnits,intentUnits:i?.quantityUnits,finalQty:os.map(x=>x.quantity),rawSide:raw?.tradeSide,normalizedSide:side,finalSide:os.map(x=>x.side)}}}
const traces=runs.filter(x=>x.startedAt>=since).map(trace);
const summary={asOf:now,identity,cohorts:{h1:cohort(now-3600000),h6:cohort(now-21600000),h24:cohort(now-86400000),currentBuild:cohort(since)},positions:{count:positions.length,notional:stats(positions.map(x=>x.quantity*x.markPrice)),margin:stats(positions.map(x=>x.quantity*x.markPrice/x.leverage)),agesHours:stats(positions.filter(x=>x.openedAt>0).map(x=>(now-x.openedAt)/3600000)),unknownOpenedAt:positions.filter(x=>!(x.openedAt>0)).length,bySide:countBy(positions,x=>x.side)},runs:countBy(runs,x=>x.decision),newRuns:countBy(traces,x=>x.normalized?.decision??x.status),newOrders:orders.filter(x=>x.createdAt>=since).length,events:countBy(events,x=>x.type),blockers:countBy(events.filter(x=>/BLOCKED|REJECTED/.test(x.type)),x=>x.payload?.reason??JSON.stringify(x.payload?.reasons)),closedNet:stats(records.filter(x=>x.status==='CLOSED'&&x.closedAt>=now-86400000&&!x.duplicateOf).map(x=>x.netPnlUsd??x.netPnl))};
fs.writeFileSync(out+'/audit-summary.json',JSON.stringify(summary,null,2));fs.writeFileSync(out+'/traces-current.json',JSON.stringify(traces,null,2));fs.writeFileSync(out+'/audit-data.json',JSON.stringify({asOf:now,identity,orders,intents,plans,reserves,positions,fills,records,runs,events}));console.log(JSON.stringify(summary,null,2));db.close();
